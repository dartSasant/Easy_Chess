import { useState } from "react";
import { View } from "react-native";

import { loadPlayerName, savePlayerName } from "@/core/storage";
import { CampaignFlow } from "@/ui/campaign/campaignFlow";
import { HomeScreen } from "@/ui/home/homeScreen";
import { NameScreen } from "@/ui/home/nameScreen";
import { LobbyScreen } from "@/ui/lobby/lobbyScreen";
import { MascotHost } from "@/ui/mascot/mascotHost";
import { useMascotBlockedWhile } from "@/ui/mascot/mascotStore";

type Screen = "home" | "name" | "lobby" | "campaign";

export function AppFlow() {
  "use no memo";
  const [name, setName] = useState<string | null>(() => loadPlayerName());
  const [screen, setScreen] = useState<Screen>(name ? "home" : "name");

  // The mascot stays away while you type your name. Games block it themselves.
  const onNameScreen = screen === "name" || !name;
  useMascotBlockedWhile(onNameScreen, "name");

  let content;
  if (screen === "name" || !name) {
    content = (
      <NameScreen
        initial={name ?? ""}
        onSave={(value) => {
          savePlayerName(value);
          setName(value);
          setScreen("home");
        }}
        onCancel={name ? () => setScreen("home") : undefined}
      />
    );
  } else if (screen === "lobby") {
    content = <LobbyScreen playerName={name} onExit={() => setScreen("home")} />;
  } else if (screen === "campaign") {
    content = <CampaignFlow onExit={() => setScreen("home")} />;
  } else {
    content = (
      <HomeScreen
        playerName={name}
        onFindPlayer={() => setScreen("lobby")}
        onPlayBot={() => setScreen("campaign")}
        onChangeName={() => setScreen("name")}
      />
    );
  }

  return (
    <View style={{ flex: 1 }}>
      {content}
      {/* One mascot above every screen. */}
      <MascotHost />
    </View>
  );
}

import type { BleTransport } from "@/core/transport";

const FILES = ["a", "b", "c", "d", "e", "f", "g", "h"];
const RANKS = ["1", "2", "3", "4", "5", "6", "7", "8"];
const START_COUNTS = { p: 8, n: 2, b: 2, r: 2, q: 1 };
const CAPTURE_ORDER = ["q", "r", "b", "n", "p"] as const;
const PROMOTION_CHOICES: PromotionPiece[] = ["q", "r", "b", "n"];

const DRAW_TEXT = {
  stalemate: "stalemate",
  "insufficient-material": "insufficient material",
  "threefold-repetition": "threefold repetition",
  "fifty-move-rule": "the fifty-move rule",
} as const;

/** Pieces of the opposite color that are missing from the board. */
function capturedBy(session: GameSession, capturer: Color): PieceType[] {
  const victim: Color = capturer === "w" ? "b" : "w";
  const onBoard = { p: 0, n: 0, b: 0, r: 0, q: 0 };

  for (const f of FILES) {
    for (const r of RANKS) {
      const piece = session.pieceAt(`${f}${r}`);
      if (piece && piece.color === victim && piece.type !== "k") {
        onBoard[piece.type] += 1;
      }
    }
  }

  const result: PieceType[] = [];
  for (const type of CAPTURE_ORDER) {
    const missing = Math.max(0, START_COUNTS[type] - onBoard[type]);
    for (let i = 0; i < missing; i++) result.push(type);
  }
  return result;
}

function statusText(session: GameSession): string {
  const status = session.status();
  if (status.kind === "checkmate") {
    return `Checkmate. ${status.winner === "w" ? "White" : "Black"} wins`;
  }
  if (status.kind === "draw") {
    return `Draw by ${DRAW_TEXT[status.reason]}`;
  }
  const side = session.turn() === "w" ? "White" : "Black";
  return status.check ? `${side} to move, check` : `${side} to move`;
}

/**
 * Sound + haptic for a move that was just applied.
 * The move text (SAN) tells us about captures ("x") and checks ("+").
 * If the game just ended, the game-end sound plays instead.
 */
function playMoveSound(session: GameSession, san: string, remote = false) {
  if (session.status().kind !== "playing") return;
  moveFeedback({ capture: san.includes("x"), check: san.includes("+"), remote });
}

type Result = "win" | "loss" | "draw";

type Props = {
  localColor?: Color;
  /** Test mode: one phone plays both sides. Turn off for real games. */
  testMode?: boolean;
  /** Bluetooth link to the opponent. Without it, the screen has no network. */
  transport?: BleTransport;
  opponentName?: string;
  /** Bluetooth: both players agreed to a rematch. Bot: "Play again". The parent starts a new game. */
  onRematch?: () => void;
  /** Go back to the previous screen. */
  onLeave?: () => void;
  /** Play against this bot instead of a person. */
  bot?: Bot;
  /** Called once when a game against a bot ends. */
  onResult?: (result: Result) => void;
  /** Offered after a win against a bot. */
  onNextBot?: () => void;
};

export function GameScreen({
  localColor = "w",
  testMode = false,
  transport,
  opponentName = "Opponent",
  onRematch,
  onLeave,
  bot,
  onResult,
  onNextBot,
}: Props) {
  "use no memo";
  const theme = useTheme();
  const [session] = useState(() => new GameSession(localColor));
  const [tick, setTick] = useState(0);
  const [selected, setSelected] = useState<Square | null>(null);
  const [targets, setTargets] = useState<Square[]>([]);
  const [pending, setPending] = useState<{ from: Square; to: Square } | null>(null);
  const [ended, setEnded] = useState<string | null>(null);
  const [endedResult, setEndedResult] = useState<Result | null>(null);
  const [drawState, setDrawState] = useState<"none" | "sent" | "received">("none");
  const [rematch, setRematch] = useState<"none" | "sent" | "received">("none");
  const [linkClosed, setLinkClosed] = useState(false);
  const [resignAsk, setResignAsk] = useState(false);
  const [notice, setNotice] = useState<{ title: string; message: string } | null>(null);
  const [disconnectMsg, setDisconnectMsg] = useState<string | null>(null);
  const reported = useRef(false);
  const endSoundPlayed = useRef(false);

  const online = !!transport && !testMode;
  const vsBot = !!bot && !testMode;
  const opponentColor: Color = localColor === "w" ? "b" : "w";
  const oppName = bot ? bot.name : opponentName;
  const oppLabel = bot ? `${bot.name} · Level ${bot.level}` : opponentName;
  const status = session.status();
  const isOver = status.kind !== "playing" || ended !== null;
  const thinking = vsBot && !isOver && !session.isMyTurn();
  const clock = useGameClock(session.turn(), !isOver);

  const outcome: Result | null =
    status.kind === "checkmate"
      ? status.winner === localColor
        ? "win"
        : "loss"
      : status.kind === "draw"
        ? "draw"
        : endedResult;

  // Bot speech bubbles and reactions (campaign games only).
  const chat = useBotChatter({
    level: bot?.level ?? 1,
    enabled: vsBot,
    moves: session.history(),
    localColor,
    outcome,
  });

  /** Ends the game by resignation, agreement or disconnect. The first ending wins. */
  const finish = (text: string, result: Result | null) => {
    setEnded((prev) => prev ?? text);
    setEndedResult((prev) => prev ?? result);
  };

  // Sound: silence the menu ambience while this screen is open, and play a
  // short "new game" sound when it opens.
  useEffect(() => {
    enterGame();
    feedback.newGame();
    return () => leaveGame();
  }, []);

  // Sound: play the win / lose / draw sound once when the game ends.
  useEffect(() => {
    if (!outcome || endSoundPlayed.current) return;
    endSoundPlayed.current = true;
    gameEndFeedback(outcome === "loss" ? "lose" : outcome);
  }, [outcome]);

  // The mascot stays away while a game is in progress, then peeks in with a line when it ends.
  useMascotBlockedWhile(!isOver);
  const mascotSpoke = useRef(false);
  useEffect(() => {
    if (!outcome || mascotSpoke.current) return;
    mascotSpoke.current = true;
    requestMascotPeek({ line: pickMascotLine(outcome), delayMs: 1200 });
  }, [outcome]);

  // Tell the campaign once when a bot game is over.
  useEffect(() => {
    if (!vsBot || !onResult || !outcome || reported.current) return;
    reported.current = true;
    onResult(outcome);
  }, [vsBot, onResult, outcome]);

  // The bot moves whenever it is its turn.
  useEffect(() => {
    if (!bot || testMode || ended !== null) return;
    if (session.status().kind !== "playing" || session.isMyTurn()) return;

    let cancelled = false;
    bot
      .chooseMove(session.fen())
      .then((move) => {
        if (cancelled || !move) return;
        const result = session.applyRemote(move);
        if (result.ok) {
          playMoveSound(session, result.move.san, true);
          setTick((t) => t + 1);
        } else Alert.alert("Bot error", result.error);
      })
      .catch(() => {
        // The bot failed. Nothing to apply, so the player simply keeps waiting.
      });

    return () => {
      cancelled = true;
    };
  }, [bot, testMode, ended, tick, session]);

  // Bluetooth messages and link events.
  useEffect(() => {
    if (!transport || testMode) return;

    const offMessage = transport.onMessage((msg) => {
      switch (msg.t) {
        case "move": {
          const result = session.applyRemote({
            from: msg.from,
            to: msg.to,
            promotion: msg.promotion,
          });
          if (result.ok) {
            playMoveSound(session, result.move.san, true);
            setDrawState("none");
            setTick((t) => t + 1);
          } else {
            feedback.error();
            Alert.alert("Opponent sent a bad move", result.error);
          }
          break;
        }
        case "resign":
          finish(`${opponentName} resigned. You win`, "win");
          break;
        case "drawOffer":
          feedback.drawOffer();
          setDrawState("received");
          break;
        case "drawAccept":
          setDrawState("none");
          finish("Draw agreed", "draw");
          break;
        case "drawDecline":
          setDrawState("none");
          setNotice({ title: "Draw declined", message: `${opponentName} declined your offer.` });
          break;
        case "rematchOffer":
          // If we offered at the same moment, show their offer so one of us can accept it.
          feedback.popup();
          setRematch("received");
          break;
        case "rematchAccept":
          // Ignored when we have no open offer, e.g. both accepted at once and we already restarted.
          if (rematch !== "none") onRematch?.();
          break;
        case "rematchDecline":
          feedback.popup();
          setRematch("none");
          Alert.alert("Rematch declined", `${opponentName} does not want a rematch.`);
          break;
        case "leave":
          setLinkClosed(true);
          setRematch("none");
          if (session.status().kind === "playing") {
            setDisconnectMsg(`${opponentName} left the game.`);
            finish(`${opponentName} left the game`, null);
          }
          break;
      }
    });

    const offEvent = transport.onEvent((e) => {
      if (e.kind === "closed") {
        setLinkClosed(true);
        setRematch("none");
        if (session.status().kind === "playing") {
          setDisconnectMsg(`Connection lost: ${e.reason}`);
          finish(e.reason, null);
        }
      }
    });

    return () => {
      offMessage();
      offEvent();
    };
  }, [transport, testMode, session, opponentName, rematch, onRematch]);

  const select = (square: Square | null) => {
    setSelected(square);
    setTargets(square ? session.legalMovesFrom(square) : []);
  };

  const play = async (input: MoveInput) => {
    const result = session.isMyTurn() ? session.moveLocal(input) : session.applyRemote(input);

    setPending(null);
    select(null);
    if (!result.ok) {
      feedback.error();
      Alert.alert("Move not allowed", result.error);
      return;
    }
    playMoveSound(session, result.move.san);
    setTick((t) => t + 1);

    if (online && transport) {
      setDrawState("none");
      const sent = await transport.send({
        t: "move",
        from: input.from,
        to: input.to,
        promotion: input.promotion,
        ply: result.move.ply,
      });
      if (!sent) Alert.alert("Move not sent", "The opponent may have disconnected.");
    }
  };

  const onSquarePress = (square: Square) => {
    if (isOver) return;
    if (!testMode && !session.isMyTurn()) return;

    if (selected && targets.includes(square)) {
      if (session.isPromotionMove(selected, square)) {
        setPending({ from: selected, to: square });
      } else {
        void play({ from: selected, to: square });
      }
      return;
    }

    const piece = session.pieceAt(square);
    const mine = !!piece && piece.color === session.turn();
    if (mine) haptics.tap(); // light tick when you pick up a piece
    select(mine ? square : null);
  };

  const comingSoon = (what: string) =>
    Alert.alert(what, "This needs the Bluetooth connection or a bot.");

  const resign = () => {
    feedback.tap();
    if (!vsBot && !(online && transport)) return comingSoon("Resign");
    setResignAsk(true);
  };

  const confirmResign = () => {
    setResignAsk(false);
    if (online && transport) void transport.send({ t: "resign" });
    finish("You resigned", "loss");
  };

  const offerDraw = () => {
    feedback.tap();
    if (vsBot && bot) {
      if (bot.acceptsDraw(session.fen(), opponentColor)) {
        finish("Draw agreed", "draw");
      } else {
        setNotice({ title: "Draw declined", message: `${bot.name} wants to play on.` });
      }
      return;
    }
    if (!online || !transport) return comingSoon("Offer draw");
    void transport.send({ t: "drawOffer" });
    setDrawState("sent");
  };

  const answerDraw = (accept: boolean) => {
    if (!transport) return;
    feedback.tap();
    void transport.send({ t: accept ? "drawAccept" : "drawDecline" });
    setDrawState("none");
    if (accept) finish("Draw agreed", "draw");
  };

  const offerRematch = () => {
    if (!transport) return;
    feedback.tap();
    void transport.send({ t: "rematchOffer" });
    setRematch("sent");
  };

  const answerRematch = (accept: boolean) => {
    if (!transport) return;
    feedback.tap();
    void transport.send({ t: accept ? "rematchAccept" : "rematchDecline" });
    if (accept) onRematch?.();
    else setRematch("none");
  };

  const leave = async () => {
    feedback.nav();
    if (online && transport && !linkClosed) {
      await transport.send({ t: "leave" }); // tell the other phone before Bluetooth stops
    }
    onLeave?.();
  };

  const headline = ended ?? (thinking ? `${oppLabel} is thinking...` : statusText(session));

  return (
    <SafeAreaView className="flex-1 bg-canvas">
      <View className="flex-1 justify-center">
        <PlayerCard
          name={oppName}
          subtitle={bot ? `Level ${bot.level}` : undefined}
          captured={capturedBy(session, opponentColor)}
          capturedColor={localColor}
          seconds={clock[opponentColor]}
          active={!isOver && session.turn() === opponentColor}
        />

        {/* Fixed-height slot, so the layout does not jump when the bubble appears. */}
        {vsBot && bot ? (
          <View style={{ height: 48, justifyContent: "center" }}>
            {chat ? <BotBubble level={bot.level} chat={chat} /> : null}
          </View>
        ) : null}

        <View className="items-center">
          <Board
            pieceAt={(sq) => session.pieceAt(sq)}
            orientation={localColor}
            selected={selected}
            legalTargets={targets}
            showCoordinates
            onSquarePress={onSquarePress}
            disabled={isOver}
          />
        </View>

        <View className="mt-2">
          <MoveList moves={session.history()} />
        </View>

        <Text className="mt-2 text-center text-base font-bold text-ink">{headline}</Text>

        {isOver && (
          <View className="mt-2 items-center">
            {vsBot && outcome === "win" ? <Trophy /> : null}

            {online && rematch === "received" && (
              <View className="flex-row items-center justify-center">
                <Text className="mr-3 text-ink">{opponentName} wants a rematch</Text>
                <Pressable
                  onPress={() => answerRematch(true)}
                  className="mr-2 rounded-lg bg-accent px-3 py-2"
                >
                  <Text className="font-bold text-ink">Accept</Text>
                </Pressable>
                <Pressable
                  onPress={() => answerRematch(false)}
                  className="rounded-lg border border-danger px-3 py-2"
                >
                  <Text className="font-bold text-danger">Decline</Text>
                </Pressable>
              </View>
            )}

            {online && linkClosed && <Text className="text-ink">Rematch is not available.</Text>}

            <View className="mt-2 flex-row">
              {vsBot && outcome === "win" && onNextBot && (
                <Pressable
                  onPress={() => {
                    feedback.nav();
                    onNextBot();
                  }}
                  className="mr-2 rounded-lg bg-accent px-4 py-3"
                >
                  <Text className="font-bold text-ink">Next opponent</Text>
                </Pressable>
              )}
              {vsBot && onRematch && (
                <Pressable
                  onPress={() => {
                    feedback.nav();
                    onRematch();
                  }}
                  className="mr-2 rounded-lg bg-accent px-4 py-3"
                >
                  <Text className="font-bold text-ink">Play again</Text>
                </Pressable>
              )}
              {online && !linkClosed && rematch !== "received" && (
                <Pressable
                  onPress={offerRematch}
                  disabled={rematch === "sent"}
                  className="mr-2 rounded-lg bg-accent px-4 py-3"
                >
                  <Text className="font-bold text-ink">
                    {rematch === "sent" ? "Waiting..." : "Rematch"}
                  </Text>
                </Pressable>
              )}
              {onLeave && (
                <Pressable onPress={() => void leave()} className="rounded-lg bg-bar px-4 py-3">
                  <Text className="font-bold text-ink">
                    {vsBot ? "Back to bots" : "Back to home"}
                  </Text>
                </Pressable>
              )}
            </View>
          </View>
        )}

        <PlayerCard
          name="You"
          captured={capturedBy(session, localColor)}
          capturedColor={opponentColor}
          seconds={clock[localColor]}
          active={!isOver && session.turn() === localColor}
        >
          <Pressable
            onPress={resign}
            disabled={isOver}
            className="mr-2 rounded-lg border border-danger px-3 py-2"
          >
            <Text className="font-bold text-danger">Resign</Text>
          </Pressable>
          <Pressable
            onPress={offerDraw}
            disabled={isOver || drawState === "sent"}
            className="rounded-lg bg-accent px-3 py-2"
          >
            <Text className="font-bold text-ink">
              {drawState === "sent" ? "Offered" : "Offer draw"}
            </Text>
          </Pressable>
        </PlayerCard>
      </View>

      {/* Promotion choice */}
      <Modal
        transparent
        animationType="fade"
        visible={pending !== null}
        onRequestClose={() => setPending(null)}
      >
        <Pressable
          className="flex-1 items-center justify-center bg-ink/60"
          onPress={() => setPending(null)}
        >
          <View className="flex-row rounded-2xl bg-bar p-3">
            {PROMOTION_CHOICES.map((type) => (
              <Pressable
                key={type}
                accessibilityLabel={`Promote to ${type}`}
                onPress={() => pending && void play({ ...pending, promotion: type })}
                className="mx-1 rounded-xl bg-canvas"
              >
                <PieceView
                  piece={{ color: session.turn(), type }}
                  size={56}
                  styleId={theme.pieceStyle}
                />
              </Pressable>
            ))}
          </View>
        </Pressable>
      </Modal>

      {/* Resign confirmation */}
      <GamePopup
        visible={resignAsk}
        title="Are you sure?"
        message="You will lose this game."
        onClose={() => setResignAsk(false)}
        actions={[
          { label: "Cancel", tone: "neutral", onPress: () => setResignAsk(false) },
          { label: "Resign", tone: "danger", onPress: confirmResign },
        ]}
      />

      {/* Draw offered to you */}
      <GamePopup
        visible={drawState === "received" && !isOver}
        title="Draw offered"
        message={`${opponentName} offers a draw.`}
        sound={false}
        actions={[
          { label: "Decline", tone: "danger", onPress: () => answerDraw(false) },
          { label: "Accept", tone: "primary", onPress: () => answerDraw(true) },
        ]}
      />

      {/* Draw declined and other short notices */}
      <GamePopup
        visible={notice !== null}
        title={notice?.title ?? ""}
        message={notice?.message}
        onClose={() => setNotice(null)}
        actions={[{ label: "OK", tone: "primary", onPress: () => setNotice(null) }]}
      />

      {/* Opponent left or the link dropped */}
      <GamePopup
        visible={disconnectMsg !== null}
        title="Opponent disconnected"
        message={disconnectMsg ?? ""}
        actions={[
          { label: "Stay", tone: "neutral", onPress: () => setDisconnectMsg(null) },
          ...(onLeave
            ? [{ label: "Leave", tone: "primary" as const, onPress: () => void leave() }]
            : []),
        ]}
      />

      {/* Confetti falls over everything after a win against a bot */}
      {vsBot && outcome === "win" ? <Confetti /> : null}
    </SafeAreaView>
  );
}

import { useEffect, useRef, useState } from "react";
import { Animated, PanResponder, View } from "react-native";
import { useSafeAreaInsets } from "react-native-safe-area-context";

import { feedback } from "@/core/feedback";

const REST = 34; // string length when idle
const KNOB = 22; // knob size
const MAX_PULL = 120; // the string stretches at most this far
const TRIGGER = 70; // pull past this and letting go goes back
const LEFT = 20;

type Props = { onPull: () => void };

/** A string with a knob hanging from the top-left. Pull it down to go back. */
export function PullString({ onPull }: Props) {
  "use no memo";
  const insets = useSafeAreaInsets();
  const [pull] = useState(() => new Animated.Value(0));
  const onPullRef = useRef(onPull);
  const crossed = useRef(false);

  useEffect(() => {
    onPullRef.current = onPull;
  });

  const springBack = () =>
    Animated.spring(pull, {
      toValue: 0,
      friction: 4,
      tension: 140,
      useNativeDriver: false, // the string's height cannot use the native driver
    }).start();

  /* eslint-disable react-hooks/refs -- these refs are only read inside gesture handlers, never during render */
  const [pan] = useState(() =>
    PanResponder.create({
      onStartShouldSetPanResponder: () => true,
      onPanResponderGrant: () => {
        pull.stopAnimation();
        crossed.current = false;
      },
      onPanResponderMove: (_, g) => {
        const d = Math.max(0, Math.min(MAX_PULL, g.dy));
        pull.setValue(d);
        // A small tick the moment the pull is enough to go back.
        if (d >= TRIGGER && !crossed.current) {
          crossed.current = true;
          feedback.tap();
        } else if (d < TRIGGER) {
          crossed.current = false;
        }
      },
      onPanResponderRelease: (_, g) => {
        const fired = g.dy >= TRIGGER || (g.vy > 1.2 && g.dy > 20);
        springBack();
        if (fired) {
          feedback.nav();
          onPullRef.current();
        }
      },
      onPanResponderTerminate: springBack,
    }),
  );
  /* eslint-enable react-hooks/refs */

  const height = pull.interpolate({
    inputRange: [0, MAX_PULL],
    outputRange: [REST, REST + MAX_PULL],
    extrapolate: "clamp",
  });

  return (
    <View
      {...pan.panHandlers}
      accessible
      accessibilityRole="button"
      accessibilityLabel="Pull to go back"
      accessibilityActions={[{ name: "activate" }]}
      onAccessibilityAction={() => onPullRef.current()}
      style={{
        position: "absolute",
        top: insets.top,
        left: LEFT,
        width: 44, // wider than the string so it is easy to grab
        alignItems: "center",
      }}
    >
      <Animated.View style={{ width: 2, height }}>
        <View className="flex-1 bg-ink" />
      </Animated.View>
      <View
        className="rounded-full bg-accent"
        style={{ width: KNOB, height: KNOB }}

}

  top: "180deg",

  right: "-90deg",

  top: 180,
  left: 90,

  top: { x: 0, y: -OUT_PX },
  left: { x: -OUT_PX, y: 0 },
  right: { x: OUT_PX, y: 0 },
};

const rand = (min: number, max: number) => min + Math.random() * (max - min);

/** A spot on the given edge. */
function placeOnEdge(edge: Edge, w: number, h: number): Place {
  if (edge === "top") {
    // Stay near the middle: the clock and battery sit in the corners of the status bar.
    const spread = Math.min(w * 0.08, 24);
    return { edge, along: (w - SIZE) / 2 + rand(-spread, spread) };
  }
  if (edge === "bottom") {
    return { edge, along: rand(MARGIN, Math.max(MARGIN, w - SIZE - MARGIN)) };
  }
  // Left / right: keep away from the very top and bottom of the screen.
  const min = h * 0.18;
  return { edge, along: rand(min, Math.max(min, h * 0.82 - SIZE)) };
}

/** A random edge (never the same twice in a row) and a spot on it. */
function pickPlace(previous: Edge | null, w: number, h: number): Place {
  const options = EDGES.filter((e) => e !== previous);
  const edge = options[Math.floor(Math.random() * options.length)];
  return placeOnEdge(edge, w, h);
}

/** Where the mascot's square sits once it has slid in. */
function boxFor(p: Place, w: number, h: number): { left: number; top: number } {
  switch (p.edge) {
    case "bottom":
      return { left: p.along, top: h - VISIBLE_PX };
    case "top":
      return { left: p.along, top: -(SIZE - VISIBLE_PX) };
    case "left":
      return { left: -(SIZE - VISIBLE_PX), top: p.along };
    case "right":
      return { left: w - VISIBLE_PX, top: p.along };
  }
}

/** The flight from one landing spot to the other: a bowed arc plus a turn in the air. */
function jumpMotion(
  t: Animated.Value,
  from: Place,
  to: Place,
  w: number,
  h: number,
) {
  const a = boxFor(from, w, h);
  const b = boxFor(to, w, h);
  const vertical = from.edge === "top" || from.edge === "bottom";
  // Bow toward the middle of the screen so the arc never leaves it.
  const side = (a.left + b.left) / 2 + SIZE / 2 < w / 2 ? 1 : -1;

  const steps = [0, 0.25, 0.5, 0.75, 1];
  const arc = (s: number) => JUMP_ARC_PX * 4 * s * (1 - s);
  const xs = steps.map(
    (s) => a.left + (b.left - a.left) * s + (vertical ? side * arc(s) : 0),
  );
  const ys = steps.map(
    (s) => a.top + (b.top - a.top) * s + (vertical ? 0 : -arc(s)),
  );

  const startDeg = ROTATE_DEG[from.edge];
  // Top to bottom keeps turning the same way and lands upright (360 = 0).
  const endDeg = from.edge === "top" ? 360 : ROTATE_DEG[to.edge];

  return {
    translateX: t.interpolate({ inputRange: steps, outputRange: xs }),
    translateY: t.interpolate({ inputRange: steps, outputRange: ys }),
    rotate: t.interpolate({
      inputRange: [0, 1],
      outputRange: [`${startDeg}deg`, `${endDeg}deg`],
    }),
  };
}

    let jumpTimer: ReturnType<typeof setTimeout> | undefined;

        friction: 6,
        tension: 90,
        useNativeDriver: true,

      clearTimeout(nextTimer);
      nextTimer = setTimeout(() => show("peek"), rand(range[0], range[1]));

      clearTimeout(jumpTimer);
      if (!jumping) return;
      jumping = false;
      jumpT.stopAnimation();
      setJump(null);
    }

    function hide() {
      clearTimeout(hideTimer);
      if (!visible) return;
      if (jumping) {
        // Never hide while it is in the air: try again right after landing.
        hideTimer = setTimeout(hide, 250);
        return;
      }
      clearTimeout(jumpTimer);
      Animated.timing(progress, {
        toValue: 0,
        duration: 320,
        easing: Easing.in(Easing.quad),
        useNativeDriver: true,
      }).start(({ finished }) => {
        if (!alive || !finished) return; // interrupted by a new visit
        visible = false;
        current = null;
        setPeek(null);
        scheduleNext(NEXT_PEEK_MS);
      });
    }

    function holdThenHide(ms: number) {
      clearTimeout(hideTimer);
      if (menuOpenRef.current) return; // it stays while the theme menu is open
      hideTimer = setTimeout(hide, ms);
    }

    /** Leap to the opposite edge. No stop in the middle. */
    function jumpAcross() {
      if (!alive || !visible || jumping || !current) return;
      if (menuOpenRef.current || draggingRef.current || isMascotBlocked()) return;
      const { w, h } = sizeRef.current;
      if (w === 0 || h === 0) return;

      const from = current.place;
      const to = placeOnEdge(OPPOSITE[from.edge], w, h);
      jumping = true;
      clearTimeout(hideTimer); // it must not hide while in the air
      jumpT.setValue(0);
      setJump({ from, to });
      haptics.pop();

      Animated.timing(jumpT, {
        toValue: 1,
        duration: JUMP_MS,
        easing: Easing.inOut(Easing.cubic),
        useNativeDriver: true,
      }).start(({ finished }) => {
        if (!alive || !finished || !jumping) return;
        jumping = false;
        lastEdge = to.edge;
        const line = current ? current.line : "";
        current = { place: to, line };
        // Landed: the normal peeking mascot takes over at exactly this spot.
        setPeek({ place: to, line });
        setJump(null);
        haptics.peek();
        holdThenHide(3500);
      });
    }

    function show(kind: MascotLineKind, forcedLine?: string) {
      if (!alive || isMascotBlocked()) return;
      const { w, h } = sizeRef.current;
      if (w === 0 || h === 0) {
        scheduleNext([2000, 4000]); // not measured yet, try again soon
        return;
      }
      clearTimeout(nextTimer);

      const line = forcedLine ?? pickMascotLine(kind);
      const hold = Math.min(12000, 6000 + line.length * 60);

      if (visible) {
        // Already peeking: new line, stay a bit longer.
        if (current) current = { ...current, line };
        setPeek((p) => (p ? { ...p, line } : p));
        haptics.bubble();
        progress.stopAnimation();
        springIn();
        holdThenHide(hold);
        return;
      }

      const place = pickPlace(lastEdge, w, h);
      lastEdge = place.edge;
      current = { place, line };
      visible = true;
      progress.stopAnimation();
      progress.setValue(0);
      drag.setValue(0);
      setPeek({ place, line });
      haptics.peek();
      springIn();
      holdThenHide(hold);

      // Only the random visits jump, never shake or game-result visits.
      if (kind === "peek" && forcedLine === undefined && Math.random() < JUMP_CHANCE) {
        clearTimeout(jumpTimer);
        jumpTimer = setTimeout(jumpAcross, rand(JUMP_AFTER_MS[0], JUMP_AFTER_MS[1]));
      }
    }

    function handlePending() {
      const p = takePendingPeek();
      if (!p) return;
      const t = setTimeout(() => {
        requestTimers.delete(t);
        show("peek", p.line);
      }, p.waitMs);
      requestTimers.add(t);
    }

    ctl.current = {
      show,

        clearTimeout(hideTimer);

import { useEffect, useSyncExternalStore } from "react";

/**
 * Tiny shared store between the game screens and the mascot overlay.
 *  - "blockers": screens that must not be interrupted (a game in progress).
 *  - "peek requests": a screen asks the mascot to peek in with a specific line.
 */

// ---- blockers -------------------------------------------------------------

const blockers = new Set<string>();
const blockListeners = new Set<() => void>();
let blockedNow = false;

export function isMascotBlocked(): boolean {
  return blockedNow;
}

export function setMascotBlocked(key: string, on: boolean) {
  if (blockers.has(key) === on) return;
  if (on) blockers.add(key);
  else blockers.delete(key);
  blockedNow = blockers.size > 0;
  blockListeners.forEach((l) => l());
}

function subscribeBlocked(listener: () => void) {
  blockListeners.add(listener);
  return () => {
    blockListeners.delete(listener);
  };
}

/** True while any screen blocks the mascot. */
export function useMascotBlocked(): boolean {
  return useSyncExternalStore(subscribeBlocked, isMascotBlocked, isMascotBlocked);
}

/** Blocks the mascot while `active` is true and releases it when the screen goes away. */
export function useMascotBlockedWhile(active: boolean, key = "game") {
  useEffect(() => {
    setMascotBlocked(key, active);
    return () => setMascotBlocked(key, false);
  }, [active, key]);
}

// ---- peek requests ----------------------------------------------------------

export type PeekRequest = {
  /** What the mascot says. Leave out for a random line. */
  line?: string;
  /** Wait this long before peeking, e.g. so the win sound can play first. */
  delayMs?: number;
};

const PENDING_TTL_MS = 5000;
let pending: { line?: string; madeAt: number; readyAt: number } | null = null;
const requestListeners = new Set<() => void>();

/**
 * Ask the mascot to peek in. If the mascot is blocked or switched off right now, the request
 * waits a few seconds and is then dropped, so an old line never shows up later.
 */
export function requestMascotPeek(req: PeekRequest = {}) {
  const now = Date.now();
  pending = { line: req.line, madeAt: now, readyAt: now + (req.delayMs ?? 0) };
  requestListeners.forEach((l) => l());
}

/** Used by the mascot overlay. Returns how long to wait before showing, or null. */
export function takePendingPeek(): { line?: string; waitMs: number } | null {
  if (!pending) return null;
  const p = pending;
  pending = null;
  const now = Date.now();
  if (now - p.madeAt > PENDING_TTL_MS) return null;
  return { line: p.line, waitMs: Math.max(0, p.readyAt - now) };
}

export function onMascotPeekRequest(listener: () => void) {
  requestListeners.add(listener);
  return () => {
    requestListeners.delete(listener);
  };
}

import { useEffect, useRef } from "react";
import { AppState } from "react-native";

/** The small part of expo-sensors that we use. */
type Subscription = { remove: () => void };
type AccelerometerModule = {
  isAvailableAsync: () => Promise<boolean>;
  setUpdateInterval: (ms: number) => void;
  addListener: (cb: (d: { x: number; y: number; z: number }) => void) => Subscription;
};

/**
 * Loaded lazily and guarded: if this build does not contain the sensor module yet
 * (for example an old app binary), shake is simply off instead of crashing the app.
 */
function loadAccelerometer(): AccelerometerModule | null {
  try {
    // eslint-disable-next-line @typescript-eslint/no-require-imports
    return (require("expo-sensors") as { Accelerometer: AccelerometerModule }).Accelerometer;
  } catch {
    return null;
  }
}

// Tuning. Gravity alone reads about 1 g. A real shake easily goes past 2 g.
const THRESHOLD_G = 2.2; // how hard a jolt has to be
const MIN_GAP_MS = 140; // two readings closer than this count as the same jolt
const WINDOW_MS = 1000; // the jolts must happen within this time
const JOLTS_NEEDED = 3; // a single bump or a dropped phone is not a shake
const COOLDOWN_MS = 2500; // after a shake, ignore new ones for a bit
const UPDATE_MS = 60;

/** Calls `onShake` when the phone is shaken. Only listens while `enabled` and the app is open. */
export function useShake(enabled: boolean, onShake: () => void) {
  const handler = useRef(onShake);
  useEffect(() => {
    handler.current = onShake;
  });

  useEffect(() => {
    if (!enabled) return;
    const accel = loadAccelerometer();
    if (!accel) return;

    let sub: Subscription | null = null;
    let cancelled = false;
    let jolts = 0;
    let lastJoltAt = 0;
    let lastFiredAt = 0;

    const start = async () => {
      if (sub || cancelled) return;
      try {
        if (!(await accel.isAvailableAsync())) return;
      } catch {
        return;
      }
      if (sub || cancelled) return;
      try {
        accel.setUpdateInterval(UPDATE_MS);
        sub = accel.addListener(({ x, y, z }) => {
          const g = Math.sqrt(x * x + y * y + z * z);
          if (g < THRESHOLD_G) return;

          const now = Date.now();
          if (now - lastFiredAt < COOLDOWN_MS) return;
          if (now - lastJoltAt > WINDOW_MS) jolts = 0;
          if (jolts > 0 && now - lastJoltAt < MIN_GAP_MS) return;

          jolts += 1;
          lastJoltAt = now;
          if (jolts >= JOLTS_NEEDED) {
            jolts = 0;
            lastFiredAt = now;
            handler.current();
          }
        });
      } catch {
        sub = null;
      }
    };

    const stop = () => {
      sub?.remove();
      sub = null;
      jolts = 0;
    };

    if (AppState.currentState === "active") void start();
    const appSub = AppState.addEventListener("change", (s) => {
      if (s === "active") void start();
      else stop();
    });

    return () => {
      cancelled = true;
      appSub.remove();
      stop();
    };
  }, [enabled]);
}
